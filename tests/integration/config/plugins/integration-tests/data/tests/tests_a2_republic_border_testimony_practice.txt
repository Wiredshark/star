# Headless actual-game acceptance for the Republic Border Testimony Practice.
# Drives hidden and visible-disabled production choices through the normal
# spaceport UI, persists the selected practice, and exercises the later A1 reader.


test "A2 Republic Border Testimony Prepare"
	status partial
	description "Load a quiet Republic save and arm the integrated B2 aftermath with recovered A1 border pressure."
	sequence
		inject "Three Earthly Barges Save"
		call "Load First Savegame"
		apply
			"Intro [0]: offered" = 1
			"Intro [0]: declined" = 1
			"A2 Career Review: seen" = 1
			"B2 Republic Border Testimony Compact: aftermath seen" = 1
			"world: republic border pressure" = 0
		assert
			"B2 Republic Border Testimony Compact: aftermath seen" == 1
			"world: republic border pressure" == 0


test "A2 Republic Border Testimony Persist And Recur"
	status partial
	description "Persist the chosen route through save/reload and exercise the production recurrence mission."
	sequence
		input
			key l
		call "Save To Default Snapshot"
		call "Open Menu"
		call "Load First Savegame"
		assert
			"A2 Republic Border Testimony Practice: decided" == 1
			"A2 Republic Border Testimony Practice: recurrence seen" == 0
		apply
			"world: republic border pressure" = 4
		input
			key p
		input
			key "Return"
		assert
			"A2 Republic Border Testimony Practice: recurrence seen" == 1
			"A2 Republic Border Testimony Practice: Recurrence: active" == 0
		input
			key l
		input
			key p
		assert
			"A2 Republic Border Testimony Practice: recurrence seen" == 1
			"A2 Republic Border Testimony Practice: Recurrence: active" == 0

test "A2 Republic Border Testimony Local"
	status active
	description "Verify the visible trust-gated response rejects selection when trust is absent, then take the valid local route."
	sequence
		call "A2 Republic Border Testimony Prepare"
		input
			key p
		input
			key "Return"
		# With no B2 settlement and no Verran trust, the trust response remains
		# visible but disabled. Return must fail without writing route state.
		input
			key "Return"
		assert
			"A2 Republic Border Testimony Practice: decided" == 0
			"A2 Republic Border Testimony Practice: independence" == 0
		input
			key "Down"
		input
			key "Return"
		input
			key "Return"
		assert
			"A2 Republic Border Testimony Practice: decided" == 1
			"A2 Republic Border Testimony Practice: local" == 1
			"A2 Republic Border Testimony Practice: Briefing: active" == 0
		call "A2 Republic Border Testimony Persist And Recur"


test "A2 Republic Border Testimony Refusal"
	status active
	description "Exercise explicit refusal with the trust response visible-but-disabled and verify persistence without recurrence."
	sequence
		call "A2 Republic Border Testimony Prepare"
		input
			key p
		input
			key "Return"
		# Displayed choices are disabled trust, local, refusal.
		input
			key "Down"
		input
			key "Down"
		input
			key "Return"
		input
			key "Return"
		assert
			"A2 Republic Border Testimony Practice: refused" == 1
			"A2 Republic Border Testimony Practice: decided" == 0
			"A2 Republic Border Testimony Practice: recurrence seen" == 0
			"A2 Republic Border Testimony Practice: Briefing: active" == 0
		input
			key l
		call "Save To Default Snapshot"
		call "Open Menu"
		call "Load First Savegame"
		assert
			"A2 Republic Border Testimony Practice: refused" == 1
			"A2 Republic Border Testimony Practice: decided" == 0
			"A2 Republic Border Testimony Practice: recurrence seen" == 0
