# A1 world simulation: Syndicate maintenance backlog -> Southern Rim transit spillover.
#
# This bridge consumes the existing Syndicate maintenance backlog as read-only
# upstream state. Severe backlog can add one temporary unit of Southern Rim
# congestion when the player next enters a qualifying Free Worlds border system.
# A six-day latch prevents repeated crossings from turning one maintenance episode
# into runaway congestion. The accepted three-day congestion decay owns recovery.


event "ES A1: Syndicate Maintenance Transit Spillover Latch Ends"
	clear "world: syndicate maintenance transit spillover active"


mission "ES A1: Syndicate Maintenance Transit Spillover"
	invisible
	entering
	non-blocking
	repeat
	source
		government "Free Worlds"
		neighbor government "Republic" "Neutral" "Independent"
	destination "Earth"
	to offer
		not "entered system by: takeoff"
		"world: syndicate maintenance backlog" >= 4
		"world: southern rim transit congestion" < 6
		not "world: syndicate maintenance transit spillover active"
	on offer
		set "world: syndicate maintenance transit spillover active"
		"world: southern rim transit congestion" += 1
		"world: southern rim transit congestion" <?= 6
		event "ES A1: Southern Rim Transit Congestion Decay" 3 3
		event "ES A1: Syndicate Maintenance Transit Spillover Latch Ends" 6 6
		fail
