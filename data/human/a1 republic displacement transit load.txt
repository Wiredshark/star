# A1 world simulation: Republic displacement -> Southern Rim transit load.
#
# Acute civilian displacement can consume one unit of existing Southern Rim
# routing capacity on a qualifying Republic-to-Free-Worlds crossing. Displacement
# pressure remains read-only; congestion recovery remains owned by the existing
# three-day decay event. A four-day latch prevents crossing spam from multiplying
# one displacement episode into repeated traffic contributions.


event "ES A1: Republic Displacement Routing Load Ends"
	clear "world: republic displacement routing load"


mission "ES A1: Republic Displacement Transit Load"
	invisible
	entering
	non-blocking
	repeat
	source
		government "Free Worlds"
		neighbor government "Republic"
	destination "Earth"
	to offer
		not "entered system by: takeoff"
		has "previous system government: Republic"
		"world: republic displacement pressure" >= 4
		not "world: republic displacement routing load"
		"world: southern rim transit congestion" < 6
	on offer
		set "world: republic displacement routing load"
		"world: southern rim transit congestion" += 1
		"world: southern rim transit congestion" <?= 6
		event "ES A1: Southern Rim Transit Congestion Decay" 3 3
		event "ES A1: Republic Displacement Routing Load Ends" 4 4
		fail
