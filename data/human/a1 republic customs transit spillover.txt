# A1 world simulation: Republic customs scrutiny -> Southern Rim transit spillover.
#
# Elevated Republic customs scrutiny now propagates beyond Republic ports. When a
# captain under elevated scrutiny crosses directly from Republic space into Free
# Worlds territory, diversion and secondary-clearance traffic consumes one unit of
# the existing Southern Rim routing capacity. The bridge deliberately reuses the
# authoritative congestion variable and its three-day decay event rather than
# creating a parallel traffic-pressure model. A two-day latch prevents the same
# scrutiny episode from amplifying every rapid border crossing. The resulting loop
# is deterministic and bounded:
# customs scrutiny -> cross-border diversion -> transit congestion -> timed recovery.
# Old saves default both the numeric input and latch condition safely to zero/false.


event "ES A1: Republic Customs Diversion Load Ends"
	clear "world: republic customs diversion load"


mission "ES A1: Republic Customs Transit Spillover"
	invisible
	entering
	non-blocking
	repeat
	source
		government "Free Worlds"
		neighbor government "Republic"
	destination "Earth"
	to offer
		not "entered system by: takeoff"
		has "previous system government: Republic"
		"world: republic customs scrutiny" >= 3
		not "world: republic customs diversion load"
		"world: southern rim transit congestion" < 6
	on offer
		set "world: republic customs diversion load"
		"world: southern rim transit congestion" += 1
		"world: southern rim transit congestion" <?= 6
		event "ES A1: Southern Rim Transit Congestion Decay" 3 3
		event "ES A1: Republic Customs Diversion Load Ends" 2 2
		fail
